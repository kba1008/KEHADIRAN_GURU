// Google Apps Script untuk Sistem Kehadiran // LANGKAH: Copy semua kod di bawah ke Google Apps Script editor // Kemudian Deploy sebagai Web App // ===== KONFIGURASI ===== const SPREADSHEET_ID = '1iUdTt97cB2jikhJQWXua3exnzVQTps9FP4jjpSPzbOc'; const DRIVE_FOLDER_ID = '17bN3qRKyryr2sRorPkCbrPmbgRzL8A4W'; const SHEET_NAMES = { GURU: 'GURU', AKP: 'AKP', KEBERSIHAN: 'KEBERSIHAN', PELAWAT: 'PELAWAT', PENDAFTARAN: 'PENDAFTARAN' }; // ===== MAIN FUNCTIONS ===== function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SPREADSHEET_ID); switch(data.action) { case 'addRecord': return addAttendanceRecord(ss, data); case 'updateCheckout': return updateCheckoutTime(ss, data); case 'uploadImage': return uploadImageToDrive(data); case 'getRecords': return getRecords(ss, data); case 'registerUser': return registerUser(ss, data); case 'validateUser': return validateUser(ss, data); default: return createResponse(false, 'Unknown action'); } } catch(error) { Logger.log('Error: ' + error); return createResponse(false, 'Server error: ' + error); } } function doGet(e) { return ContentService .createTextOutput(JSON.stringify({status: 'ok', version: '1.0'})) .setMimeType(ContentService.MimeType.JSON); } // ===== USER MANAGEMENT FUNCTIONS ===== function registerUser(ss, data) { try { const sheet = ss.getSheetByName(SHEET_NAMES.PENDAFTARAN); if (!sheet) return createResponse(false, 'Sheet PENDAFTARAN not found'); // Check duplicates const rows = sheet.getDataRange().getValues(); // Skip header row 0 for (let i = 1; i < rows.length; i++) { const rowName = rows[i][1].toString().toUpperCase(); const rowEmail = rows[i][3].toString().toLowerCase(); if (rowName === data.nama.toUpperCase()) { return createResponse(false, 'DUPLICATE_NAME'); } if (data.email && rowEmail === data.email.toLowerCase()) { return createResponse(false, 'DUPLICATE_EMAIL'); } } // Append new user sheet.appendRow([ new Date().toISOString(), data.nama.toUpperCase(), data.kategori, data.email ? data.email.toLowerCase() : '-', data.password, 'AKTIF' ]); return createResponse(true, 'Pendaftaran berjaya'); } catch(error) { return createResponse(false, error.toString()); } } function validateUser(ss, data) { try { const sheet = ss.getSheetByName(SHEET_NAMES.PENDAFTARAN); if (!sheet) return createResponse(false, 'Sheet PENDAFTARAN not found'); const rows = sheet.getDataRange().getValues(); let isValid = false; let userData = null; for (let i = 1; i < rows.length; i++) { // For Guru: Check Email if (data.email) { if (rows[i][3].toString().toLowerCase() === data.email.toLowerCase()) { isValid = true; userData = { nama: rows[i][1], kategori: rows[i][2] }; break; } } // For Others: Check Name else if (data.nama) { if (rows[i][1].toString().toUpperCase() === data.nama.toUpperCase()) { isValid = true; userData = { nama: rows[i][1], kategori: rows[i][2] }; break; } } } if(isValid) { return createResponse(true, 'User valid', userData); } else { return createResponse(false, 'User not found in registry'); } } catch(error) { return createResponse(false, error.toString()); } } // ===== ATTENDANCE FUNCTIONS ===== function addAttendanceRecord(ss, data) { try { const kategori = data.kategori || 'UMUM'; let sheet = ss.getSheetByName(SHEET_NAMES[kategori]); if (!sheet) { sheet = ss.insertSheet(SHEET_NAMES[kategori]); sheet.appendRow(['Timestamp', 'Tarikh', 'Nama', 'Emel', 'Waktu Masuk', 'Waktu Keluar', 'Lokasi GPS', 'Device ID', 'Link Gambar', 'Tujuan', 'Status']); sheet.setFrozenRows(1); } sheet.appendRow([ new Date().toISOString(), data.tarikh || new Date().toLocaleDateString('ms-MY'), data.nama || '-', data.emel || '-', data.waktuMasuk || '-', data.waktuKeluar || '-', data.lokasiGPS || '-', data.deviceId || '-', data.linkGambar || '-', data.tujuan || '-', data.statusSync || 'synced' ]); return createResponse(true, 'Record added successfully'); } catch(error) { return createResponse(false, error.toString()); } } function updateCheckoutTime(ss, data) { try { const sheet = ss.getSheetByName(SHEET_NAMES[data.kategori]); if (!sheet) return createResponse(false, 'Sheet not found'); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][7] === data.deviceId && rows[i][4] === data.waktuMasuk) { sheet.getRange(i + 1, 6).setValue(data.waktuKeluar); return createResponse(true, 'Checkout updated'); } } return createResponse(false, 'Record not found'); } catch(error) { return createResponse(false, error.toString()); } } function uploadImageToDrive(data) { try { const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID); const today = new Date().toLocaleDateString('ms-MY'); let todayFolder = null; const folders = folder.getFolders(); while (folders.hasNext()) { const f = folders.next(); if (f.getName() === today) { todayFolder = f; break; } } if (!todayFolder) todayFolder = folder.createFolder(today); const imageBlob = Utilities.newBlob( Utilities.base64Decode(data.imageBase64), 'image/jpeg', data.nama + '_' + new Date().getTime() + '.jpg' ); const file = todayFolder.createFile(imageBlob); return createResponse(true, 'Image uploaded', {url: file.getUrl()}); } catch(error) { return createResponse(false, error.toString()); } } function getRecords(ss, data) { try { const sheet = ss.getSheetByName(SHEET_NAMES[data.kategori || 'GURU']); if (!sheet) return createResponse(false, 'Sheet not found'); const records = sheet.getDataRange().getValues(); return createResponse(true, 'Records retrieved', {records: records, count: records.length}); } catch(error) { return createResponse(false, error.toString()); } } // ===== UTILITY FUNCTIONS ===== function createResponse(success, message, data = {}) { return ContentService .createTextOutput(JSON.stringify({success, message, timestamp: new Date().toISOString(), ...data})) .setMimeType(ContentService.MimeType.JSON); } function createImageFolder() { try { DriveApp.getFolderById(DRIVE_FOLDER_ID).createFolder('GAMBAR'); } catch(e) {} } function setup() { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); Object.values(SHEET_NAMES).forEach(name => { let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); if(name === 'PENDAFTARAN') { sheet.appendRow(['Timestamp', 'Nama', 'Kategori', 'Emel', 'Kata Laluan', 'Status']); } else { sheet.appendRow(['Timestamp', 'Tarikh', 'Nama', 'Emel', 'Waktu Masuk', 'Waktu Keluar', 'Lokasi GPS', 'Device ID', 'Link Gambar', 'Tujuan', 'Status']); } sheet.setFrozenRows(1); } }); createImageFolder(); }
